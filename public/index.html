// Middleware: Rate limiting with sliding window
async function rateLimiter(req, res, next) {
    const ip = req.ip;
    const requests = await redis.incr(`rate:${ip}`);
    if (requests === 1) await redis.expire(`rate:${ip}`, 60);
    if (requests > 100) return res.status(429).json({ error: 'Rate limit exceeded' });
    next();
}

app.post('/api/auth/token', rateLimiter, (req, res) => {
    const token = jwt.sign({ sub: req.body.userId }, process.env.JWT_SECRET, { expiresIn: '1h' });
    res.json({ token, expires_in: 3600 });
});

import express from 'express';

const flagsmith = new Flagsmith({
    environmentKey: process.env.FLAGSMITH_ENVIRONMENT_KEY,
    enableEvents: true
});

export async function bootstrapService() {
    const app = express();
    
    // Flagsmith dynamic rollout middleware
    app.use(async (req, res, next) => {
        const flag = await flagsmith.getExperimentFlag('new_auth_engine', req.headers['x-user-id'] || 'anon');
        req.useNewAuthEngine = flag.enabled;
        next();
    });

    return app;
}
